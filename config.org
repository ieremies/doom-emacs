  (defun jinx--add-to-abbrev (overlay word)
    "Add abbreviation to `global-abbrev-table'.
The misspelled word is taken from OVERLAY.  WORD is the corrected word."
    (let ((abbrev (buffer-substring-no-properties
                   (overlay-start overlay)
                   (overlay-end overlay))))
      (message "Abbrev: %s -> %s" abbrev word)
      (define-abbrev global-abbrev-table abbrev word)))

(after! org
  (defun org-do-latex-and-related (limit)
    "Highlight LaTeX snippets and environments, entities and sub/superscript.
Stop at first highlighted object, if any.  Return t if some
highlighting was done, nil otherwise."
    (when (org-string-nw-p org-latex-and-related-regexp)
      (let ((latex-prefix-re (rx (or "$" "\\(" "\\[")))
	    (blank-line-re (rx (and "\n" (zero-or-more (or " " "\t")) "\n"))))
        (catch 'found
	  (while (and (< (point) limit)
		      (re-search-forward org-latex-and-related-regexp nil t))
	    (cond
             ((>= (match-beginning 0) limit)
	      (throw 'found nil))
	     ((cl-some (lambda (f)
		         (memq f '(org-code org-verbatim underline
				   org-special-keyword)))
		       (save-excursion
		         (goto-char (1+ (match-beginning 0)))
		         (face-at-point nil t))))
	     ;; Try to limit false positives.  In this case, ignore
	     ;; $$...$$, \(...\), and \[...\] LaTeX constructs if they
	     ;; contain an empty line.
	     ((save-excursion
	        (goto-char (match-beginning 0))
	        (and (looking-at-p latex-prefix-re)
		     (save-match-data
		       (re-search-forward blank-line-re (1- (match-end 0)) t)))))
	     (t
	      (let* ((offset (if (memq (char-after (1+ (match-beginning 0)))
				       '(?_ ?^))
			         1
			       0))
		     (start (+ offset (match-beginning 0)))
		     (end (match-end 0)))
	        (if (memq 'native org-highlight-latex-and-related)
		    (org-src-font-lock-fontify-block "latex" start end)
		  (font-lock-prepend-text-property start end
						   'face 'org-latex-and-related))

	        ;;<<<<<<<<<<<<<<<<<<<<<
	        ;; my code starts here
	        (when (and org-hide-emphasis-markers (< (+ start 4) end))

(defun my/generate-block (env name label)
  "Generate a block for LaTeX or plain text with optional name and label."
  (if (equal backend 'latex)
      (cond  ;; LaTeX block generation
       ((and (string-equal name "") (string-equal label ""))  ;; No name and no label
        (format "\\begin{%s}%s\\end{%s}" env contents env))
       ((string-equal name "")  ;; No name, but label present
        (format "\\begin{%s}\\label{%s}%s\\end{%s}" env label contents env))
       ((string-equal label "")  ;; Name present, no label
        (format "\\begin{%s}[%s]%s\\end{%s}" env name contents env))
       (t  ;; Both name and label present
        (format "\\begin{%s}[%s]\\label{%s}%s\\end{%s}" env name label contents env)))
    (cond  ;; Plain text block generation
     ((and (string-equal name "") (string-equal label ""))  ;; No name and no label
      (format "%s: %s" (capitalize env) contents))
     ((string-equal name "")  ;; No name, but label present
      (format "%s: %s %s" (capitalize env) label contents))
     ((string-equal label "")  ;; Name present, no label
      (format "%s: [%s] %s" (capitalize env) name contents))
     (t  ;; Both name and label present
      (format "%s: [%s] %s %s" (capitalize env) name label contents)))))
